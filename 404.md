{%- comment -%}
  2026-09-27 (P2 SEO). Before this page the site had no 404 at all, so GitHub
  Pages served its generic one: no internal links, no search, and nothing to
  hand a crawler that mistypes or follows a retired URL. This page exists to
  convert both of those into crawl paths.

  `sitemap: false` in the front matter above keeps the page out of
  sitemap.xml. That is belt and braces - jekyll-sitemap already excludes
  `doc.url != "/404.html"` and `page.name != "404.html"` - but the exclusion
  then lives with the page instead of only inside a gem, so a future change to
  the plugin cannot silently start advertising it.
{%- endcomment -%}

{%- comment -%}
  The `default` layout used to have no head-content hook, so this page carried
  its robots meta as a <meta> element in the document BODY. Browsers hoist
  such a tag into <head> while parsing, so the directive was honoured, but it
  was not where a robots meta belongs and a stricter parser could miss it.

  2026-09-27 (on-page SEO pass): `_layouts/default.html` now reads a
  front-matter `noindex` key through `_includes/seo_head.html` and emits
  `<meta name="robots" content="noindex, follow">` into the real <head>, and
  `noindex: true` above is that switch. The body-level tag is removed rather
  than left alongside it, so the directive is stated exactly once.

  `follow` is deliberate and unchanged: `noindex, nofollow` would strand every
  link below, which is the opposite of what this page is for. This page is
  also the reason the site is expected to have links into /search/ and the
  four collection indexes rather than a dead end.
{%- endcomment -%}

{%- comment -%}
  The search box above is a live lookup, not a dead form: it reads the same
  three published indexes the /search/ page reads, so it only works because
  `data/` is no longer excluded from the build (see _config.yml). It submits to
  /search/ as a plain GET as well, so with JavaScript off, or if a fetch fails,
  the reader still lands on the full search page rather than nowhere.

  Result URLs are built exactly as scripts/build_collections.py builds them -
  /papers/<slugify(title)>/, /videos/<id>/, /articles/<slug>/ - and
  collection_drift() in test_canonical_57.py fails the suite if a data row has
  no generated page, so a link from here cannot 404 while the suite is green.
  Note that papers.json's `id` is a row number (1..20), NOT a slug; the paper
  URL is derived from the title. The match is over title, excerpt, keywords,
  publisher, topics and themes, all of which are already in the published
  index. Only eight results are shown; the page points at /search/ for the rest.
{%- endcomment -%}
<script>
(function () {
  var form = document.getElementById('nf-search');
  var input = document.getElementById('nf-q');
  var out = document.getElementById('nf-results');
  if (!form || !input || !out) { return; }

  var MAX = 8;
  var INDEXES = [
    { src: '{{ "/data/papers.json" | relative_url }}', base: '{{ "/papers/" | relative_url }}', slug: 'title', meta: ['publisher_journal', 'type'] },
    { src: '{{ "/data/videos.json" | relative_url }}', base: '{{ "/videos/" | relative_url }}', slug: 'id', meta: ['format', 'tone'] },
    { src: '{{ "/data/canonical_works.json" | relative_url }}', base: '{{ "/articles/" | relative_url }}', slug: 'slug', meta: ['date', 'status'] }
  ];

  function esc(s) {
    var d = document.createElement('div');
    d.appendChild(document.createTextNode(s == null ? '' : String(s)));
    return d.innerHTML;
  }

  // Must match paper_slug() in scripts/build_collections.py.
  function slugify(title) {
    return String(title || '').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '');
  }

  function haystack(row) {
    var parts = [row.title, row.excerpt, row.slug, row.publisher_journal, row.type, row.format, row.tone, row.date, row.status];
    if (row.keywords) { parts = parts.concat(row.keywords); }
    if (row.themes) { parts = parts.concat(row.themes); }
    if (row.topics) { parts = parts.concat(row.topics); }
    return parts.join(' ').toLowerCase();
  }

  function render(rows, total) {
    if (!total) {
      out.innerHTML = '<div class="search-no-results">Nothing in the archive matches that. The four collections below are worth a look.</div>';
      return;
    }
    var html = '<div class="results-section">';
    for (var i = 0; i < rows.length; i++) {
      html += '<div class="result-item"><h4><a href="' + rows[i].href + '">' + esc(rows[i].title) + '</a></h4>';
      if (rows[i].meta) { html += '<div class="result-meta">' + esc(rows[i].meta) + '</div>'; }
      html += '</div>';
    }
    html += '</div>';
    if (total > rows.length) {
      html += '<p class="search-info">Showing the first ' + rows.length + ' of ' + total + ' matches. <a href="' + form.getAttribute('action') + '">Open the full search page</a> for the rest, with filters.</p>';
    } else {
      html += '<p class="search-info"><a href="' + form.getAttribute('action') + '">Open the full search page</a> for filters.</p>';
    }
    out.innerHTML = html;
  }

  function search(q) {
    Promise.all(INDEXES.map(function (ix) {
      return fetch(ix.src).then(function (r) { return r.json(); }).catch(function () { return []; });
    })).then(function (sets) {
      var rows = [];
      var total = 0;
      for (var i = 0; i < INDEXES.length; i++) {
        var ix = INDEXES[i];
        var set = sets[i] || [];
        for (var j = 0; j < set.length; j++) {
          var row = set[j];
          if (haystack(row).indexOf(q) === -1) { continue; }
          total++;
          if (rows.length >= MAX) { continue; }
          var key = row[ix.slug];
          var href = ix.base + encodeURIComponent(ix.slug === 'title' ? slugify(key) : String(key || '')) + '/';
          var meta = '';
          for (var k = 0; k < ix.meta.length; k++) {
            if (row[ix.meta[k]]) { meta = row[ix.meta[k]]; break; }
          }
          rows.push({ href: href, title: row.title, meta: meta });
        }
      }
      render(rows, total);
    }).catch(function () {
      out.innerHTML = '<div class="search-error">Search is unavailable here. <a href="' + form.getAttribute('action') + '">Open the search page</a>, or use the collections below.</div>';
    });
  }

  var timer = null;
  function run() {
    var q = input.value.trim().toLowerCase();
    if (q.length < 2) {
      out.innerHTML = '<p class="search-info">Type at least two characters.</p>';
      return;
    }
    out.innerHTML = '<p class="search-info">Searching&hellip;</p>';
    search(q);
  }

  input.addEventListener('input', function () {
    clearTimeout(timer);
    timer = setTimeout(run, 300);
  });
  form.addEventListener('submit', function (ev) {
    // Only intercept when the live lookup can run; otherwise let the plain GET
    // to /search/ happen, which is why the form has a real action.
    ev.preventDefault();
    run();
    input.blur();
  });
})();
</script>
